CREATE TABLE exam (
    Eid NUMBER(3),
    Subname VARCHAR2(20),
    marks NUMBER(3),
    sid NUMBER(3),
    CONSTRAINT fk_sid FOREIGN KEY (sid) REFERENCES students(sid)
);
